86## 🧩 Comprehensive Entity-Relationship (ER) Schema
87 
88The data model follows the ER diagram specified in **IEEE 7384005**:
89 
90- `STUDENT_INFO`: `(PK: id, rollNo, name, email, FK: deptId, FK: majorId, FK: classId, cgpa, totalCreditsEarned, academicStatus, promotionStatus)`
91- `TEACHER_INFO`: `(PK: id, name, department, title, email)`
92- `DEPARTMENT_INFO`: `(PK: id, code, name, head)`
93- `MAJOR_INFO`: `(PK: id, FK: deptId, name, degree)`
94- `CLASS_INFO`: `(PK: id, name, currentSemester, FK: counselorId)`
95- `COURSE_INFO`: `(PK: id, code, name, credits, type, dept, FK: instructorId)`
96- `COMPREHENSIVE_SCORE`: `(Internal [30] + Assignments [20] + Final Exam [50] = Total Score [100] ➔ Grade Point & Letter Grade)`
97- `PLACEMENT_INFO`: `(company, role, ctc, offerDate, status)`
98- `RESEARCH_AND_PROJECTS`: `(title, type, venue, guide, status)`

25function createServerInstance() {
26  return http.createServer((req, res) => {
27    let reqUrl = req.url.split('?')[0];
28    if (reqUrl === '/') reqUrl = '/index.html';
29 
30    const filePath = path.join(PUBLIC_DIR, reqUrl);
31 
32    // Prevent directory traversal
33    if (!filePath.startsWith(PUBLIC_DIR)) {
34      res.writeHead(403, { 'Content-Type': 'text/plain' });
35      res.end('Access Denied');
36      return;
37    }
38 
39    fs.stat(filePath, (err, stats) => {
40      if (err || !stats.isFile()) {
41        res.writeHead(404, { 'Content-Type': 'text/plain' });
42        res.end('404 Not Found');
43        return;
44      }
45 
46      const ext = path.extname(filePath).toLowerCase();
47      const contentType = MIME_TYPES[ext] || 'application/octet-stream';
48 
49      res.writeHead(200, {
50        'Content-Type': contentType,
51        'Cache-Control': 'no-cache, no-store, must-revalidate'
52      });
53 
54      const readStream = fs.createReadStream(filePath);
55      readStream.pipe(res);
56    });
57  });
58}

73function startServer(options = {}) {
74  const openBrowser = options.openBrowser !== false;
75  let currentPort = options.port || DEFAULT_PORT;
76  const maxPort = currentPort + 10;
77 
78  function tryListen(port) {
79    const srv = createServerInstance();
80 
81    srv.once('error', (err) => {
82      if (err.code === 'EADDRINUSE') {
83        console.warn(`Port ${port} is currently busy, attempting port ${port + 1}...`);
84        if (port < maxPort) {
85          tryListen(port + 1);
86        } else {
87          console.error(`Unable to bind to any port between ${DEFAULT_PORT} and ${maxPort}.`);
88        }
89      } else {
90        console.error('Server error:', err);
91      }
92    });
93 
94    srv.listen(port, () => {
95      const url = `http://localhost:${port}`;
96      console.log('===========================================================');
97      console.log(' Educational Institution Management System (IMS) PoC');
98      console.log(` Server is running live at: ${url}`);
99      console.log('===========================================================');
100 
101      if (openBrowser) {
102        console.log(`Opening ${url} in your default browser...`);
103        openBrowserUrl(url);
104      }
105    });
106 
107    return srv;
108  }
109 
110  return tryListen(currentPort);
111}
